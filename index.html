import requests

from typing import Dict, List, Optional

# NOAA Tides & Currents API Endpoint
NOAA_API_URL = "https://api.tidesandcurrents.noaa.gov/api/prod/datagetter"
INATURALIST_API = "https://api.inaturalist.org/v1/observations"

def fetch_noaa_tide_predictions(station_id: str, date_str: str) -> List[Dict]:
    """
    Fetch 6-minute interval tide predictions for NOAA station.
    Datum: MLLW (Mean Lower Low Water)
    """
    params = {
        "begin_date": date_str,
        "range": 24,
        "station": station_id,
        "product": "predictions",
        "datum": "MLLW",
        "time_zone": "lst_ldt",
        "units": "english",
        "format": "json"
    }
    response = requests.get(NOAA_API_URL, params=params)
    data = response.json()
    return data.get("predictions", [])

def calculate_tide_score(current_height_ft: float, min_height_ft: float) -> Dict:
    """
    Calculates Tide Pool Safety & Sightability Score based on MLLW height.
    """
    if current_height_ft <= 0.5:
        return {"status": "Optimal", "score": 95, "color": "emerald", "recommendation": "Prime tide pooling condition! Deep pockets revealed."}
    elif current_height_ft <= 1.8:
        return {"status": "Moderate", "score": 70, "color": "amber", "recommendation": "Good viewing in High to Mid intertidal zones."}
    else:
        return {"status": "High Tide - Not Recommended", "score": 25, "color": "rose", "recommendation": "Submerged tide pools. Substantial swell hazard."}

def fetch_inaturalist_intertidal_species(lat: float, lng: float, radius_km: int = 5) -> List[Dict]:
    """
    Fetch real user species observations nearby for intertidal marine life taxa (47170 = Chromista / Marine)
    """
    params = {
        "lat": lat,
        "lng": lng,
        "radius": radius_km,
        "iconic_taxa": "Mollusca,Actiniaria,Echinodermata,Crustacea",
        "per_page": 20,
        "order_by": "votes"
    }
    response = requests.get(INATURALIST_API, params=params)
    return response.json().get("results", [])
